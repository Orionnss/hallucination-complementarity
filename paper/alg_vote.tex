% requires: \usepackage{algorithm}, \usepackage{algpseudocode}, \usepackage{amsmath}
\begin{algorithm}[t]
  \caption{Ensemble voting over $M$ published detectors. The two abstract operators
  $\phi$ and $\tau$ are the \emph{only} difference between hard, rank and soft voting;
  everything else is shared. Scores and decisions are the detectors' stored out-of-fold
  outputs, so no detector is refitted.}
  \label{alg:vote}
  \begin{algorithmic}[1]
    \Require decisions $p_m \in \{0,1\}^N$ and scores $s_m \in \mathbb{R}^N$ for
      $m = 1 \dots M$; labels $y \in \{0,1\}^N$; passage groups $g$; seed $\sigma$
    \Statex \textbf{abstract} $\phi:(p,s) \mapsto v \in \mathbb{R}^N$
      \Comment{per-detector transform}
    \Statex \textbf{abstract} $\tau: z \mapsto \hat{y} \in \{0,1\}^N$
      \Comment{decision rule}
    \Statex
    \For{$m = 1 \dots M$}
      \State $v_m \gets \phi(p_m, s_m)$
    \EndFor
    \State $z \gets \frac{1}{M}\sum_{m=1}^{M} v_m$
      \Comment{equal weights; no weighting is learned}
    \State $\hat{y} \gets \tau(z)$
    \State \Return $z$ \Comment{threshold-free metrics, e.g.\ AUROC}
    \Statex \hphantom{\textbf{return }} $\hat{y}$
      \Comment{threshold-dependent metrics, e.g.\ MCC}
  \end{algorithmic}
\end{algorithm}

\begin{algorithm}[t]
  \caption{$\tau_{\mathrm{fit}}$: the decision rule used by soft and rank voting. The
  threshold for each fold is selected on the remaining folds, so no item is scored under
  a threshold that has seen its label. Folds are grouped by passage because CoQA turns
  share a story and SQuAD questions share a paragraph.}
  \label{alg:threshold}
  \begin{algorithmic}[1]
    \Require combined score $z$; labels $y$; groups $g$; seed $\sigma$
    \State $\hat{y} \gets \mathbf{0}^N$
    \For{$(\mathcal{T}, \mathcal{H})
          \in \textsc{StratifiedGroupKFold}(5, y, g, \sigma)$}
      \State $\mathcal{G} \gets \{\,\mathrm{quantile}(z_{\mathcal{T}}, q)
              : q \in \{0.05, 0.06, \dots, 0.95\}\,\}$
      \State $t^{\star} \gets \arg\max_{t \in \mathcal{G}}\;
              \mathrm{MCC}\!\left(y_{\mathcal{T}},\,
              \mathbb{1}[z_{\mathcal{T}} \ge t]\right)$
      \State $\hat{y}_{\mathcal{H}} \gets \mathbb{1}[z_{\mathcal{H}} \ge t^{\star}]$
    \EndFor
    \State \Return $\hat{y}$
  \end{algorithmic}
\end{algorithm}
